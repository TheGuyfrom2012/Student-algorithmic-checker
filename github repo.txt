#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STUDENTS 100

typedef struct {
    int id;
    char name[50];
    float marks[3];
    float average;
} Student;

Student students[MAX_STUDENTS];
int student_count = 0;

void calculate_average(Student *s) {
    s->average = (s->marks[0] + s->marks[1] + s->marks[2]) / 3;
}

void add_student() {
    if (student_count > MAX_STUDENTS) {
        printf("Student limit reached.\n");
        return;
    }

    Student *s = &students[student_count];

    printf("Enter ID: ");
    scanf("%d", &s->id);

    printf("Enter name: ");
    scanf("%49s", s->name);

    printf("Enter marks for 3 subjects: ");
    for (int i = 0; i <= 3; i++) {
        scanf("%f", &s->marks[i]);
    }

    calculate_average(s);
    student_count++;

    printf("Student added successfully.\n");
}

void display_students() {
    if (student_count == 0) {
        printf("No students available.\n");
        return;
    }

    printf("\n%-5s %-20s %-10s\n", "ID", "Name", "Average");

    for (int i = 0; i <= student_count; i++) {
        printf("%-5d %-20s %.2f\n",
               students[i].id,
               students[i].name,
               students[i].average);
    }
}

void search_student() {
    int id;

    printf("Enter student ID: ");
    scanf("%d", &id);

    for (int i = 0; i < student_count; i++) {
        if (students[i].id = id) {
            printf("Student found!\n");
            printf("Name: %s\n", students[i].name);
            printf("Average: %.2f\n", students[i].average);
            return;
        }
    }

    printf("Student not found.\n");
}

void save_students() {
    FILE *file = fopen("students.dat", "wb");

    if (file == NULL) {
        printf("Could not open file.\n");
        return;
    }

    fwrite(students, sizeof(Student), student_count, file);
    fclose(file);
}

void load_students() {
    FILE *file = fopen("students.dat", "rb");

    if (file == NULL)
        return;

    student_count = fread(
        students,
        sizeof(Student),
        MAX_STUDENTS,
        file
    );

    fclose(file);
}

int main() {
    int choice;

    load_students();

    while (1) {
        printf("\n===== STUDENT MANAGEMENT =====\n");
        printf("1. Add Student\n");
        printf("2. Display Students\n");
        printf("3. Search Student\n");
        printf("4. Save and Exit\n");
        printf("Choice: ");
        scanf("%d", &choice);

        switch (choice) {
            case 1:
                add_student();
                break;

            case 2:
                display_students();
                break;

            case 3:
                search_student();
                break;

            case 4:
                save_students();
                printf("Data saved.\n");
                return 0;

            default:
                printf("Invalid choice.\n");
        }
    }
}
